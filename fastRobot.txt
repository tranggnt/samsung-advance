	public static class Node{
		int x;
		int y;
		int dir;
		int turn;
		
		public Node(int x, int y, int dir, int turn){
			this.x = x;
			this.y = y;
			this.dir = dir;
			this.turn = turn;
		}
	}
	
	public static class Queue{
		Node[] q ;
		int front ;
		int rear;
		
		public Queue(){
			q =new Node[100000];
			front = 0 ;
			rear = -1;
		}
		
		public boolean isEmpty(){
			return rear < front;
		}
		
		public void push(Node p){
			q[++rear] = p;
		}
		
		public Node pop(){
			return q[front++];
		}
	}

	public static int bfs( int sx, int sy, int ex, int ey){
		
		Queue q = new Queue();
		
		int[][] best = new int[n+5][m+5];
		for ( int i=1; i<=n; i++){
			for ( int j=1; j<=m; j++){
				best[i][j] = Integer.MAX_VALUE;
			}
		}
		
		q.push(new Node(sx, sy, -1, 0));
		
		best[sx][sy] =0;
		
		while ( !q.isEmpty()){
			Node cur = q.pop();
				if ( cur.x == ex && cur.y==ey){
					return cur.turn;
				}
				
				for ( int k = 0; k<4; k++){
					int nx = cur.x + dx[k];
					int ny = cur.y + dy[k];
					
					if ( nx <1 || ny <1 || nx >n || ny > m){
						continue;
					}
					
					if ( a[nx][ny] == '1'){
						continue;
					}
					
					int newTurn = cur.turn;
					
					if ( newTurn != -1 && cur.dir != k){
						newTurn++;
					}
					if ( newTurn < best[nx][ny]){
						best[nx][ny] = newTurn;
						q.push(new Node ( nx, ny, k, newTurn));
					
				}
			}
		}
		return -1;
	}
