Trái đất nóng lên kéo theo mực nước biển dâng. Hòn đảo nhỏ Gonnasinka thuê bạn để dự báo trước hiểm họa này. Cho trước 1 lưới tọa độ thể hiện cao độ của đảo, hãy giúp họ tính toán xem nước biển dâng cao bao nhiêu thì hòn đảo sẽ bị chia cắt.

Input

Input gồm nhiều bộ test, mỗi bộ test bao gồm:

Dòng đầu ghi 2 số n, m là chiều dài và chiều rộng.

Sau đó là n dòng, mỗi dòng gồm m số, mỗi số cho biết độ cao của ô đó, giá trị 0 chỉ mực nước biển. Những ô giá trị 0 dọc theo đường viền và những ô số 0 liền kề những ô này chỉ mặt biển. Những ô có giá trị 0 còn lại (được bao bọc bởi các số > 0) là đất liền bên trong đảo nhưng có độ cao ngang mặt nước biển. Hòn đảo lúc đầu chưa bị chia cắt. Số n và m không lớn hơn 100 và độ cao không lớn hơn 1000.

Dòng cuối cùng của input chứa 2 số 0

Output

Với mỗi bộ test, in ra:

Case n: Island splits when ocean raises f feet. (Đảo bị chia khi nước biển dâng cao f feet)

Hoặc

Case n: Island never splits. (Đảo không bao giờ bị chia cắt)

Example

Input:

5 5

3 4 3 0 0

3 5 5 4 3

2 5 4 4 3

1 3 0 0 0

1 2 1 0 0

5 5

5 5 5 5 7

4 1 1 1 4

4 1 2 1 3

7 1 0 0 4

7 3 4 4 4

0 0

Output:

Case 1: Island never splits.

Case 2: Island splits when ocean rises 3 feet.
import java.util.*;

public class Island {

    static int n, m;
    static int[][] height;

    // 4 hướng: lên, xuống, trái, phải
    static int[] dx = {-1, 1, 0, 0};
    static int[] dy = {0, 0, -1, 1};

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int testCase = 1;

        while (true) {

            n = sc.nextInt();
            m = sc.nextInt();

            // 0 0: kết thúc
            if (n == 0 && m == 0) {
                break;
            }

            height = new int[n][m];

            int maxHeight = 0;

            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    height[i][j] = sc.nextInt();
                    maxHeight = Math.max(maxHeight, height[i][j]);
                }
            }

            int answer = -1;

            // Thử từng mức nước biển dâng
            for (int water = 1; water <= maxHeight; water++) {

                if (isSplit(water)) {
                    answer = water;
                    break;
                }
            }

            if (answer == -1) {
                System.out.println(
                    "Case " + testCase + ": Island never splits."
                );
            } else {
                System.out.println(
                    "Case " + testCase +
                    ": Island splits when ocean raises " +
                    answer + " feet."
                );
            }

            testCase++;
        }

        sc.close();
    }

    /*
     * Kiểm tra xem ở mức nước hiện tại,
     * đảo có bị chia thành nhiều phần hay không.
     */
    static boolean isSplit(int water) {

        boolean[][] visited = new boolean[n][m];

        int islandCount = 0;

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {

                // Ô này vẫn còn là đất
                if (height[i][j] > water && !visited[i][j]) {

                    islandCount++;

                    // Đã có ít nhất 2 vùng đất
                    if (islandCount >= 2) {
                        return true;
                    }

                    bfs(i, j, water, visited);
                }
            }
        }

        return false;
    }

    /*
     * BFS tìm toàn bộ một vùng đất liên thông.
     */
    static void bfs(
            int startX,
            int startY,
            int water,
            boolean[][] visited) {

        Queue<int[]> queue = new LinkedList<>();

        queue.offer(new int[]{startX, startY});
        visited[startX][startY] = true;

        while (!queue.isEmpty()) {

            int[] current = queue.poll();

            int x = current[0];
            int y = current[1];

            // Xét 4 ô xung quanh
            for (int d = 0; d < 4; d++) {

                int nx = x + dx[d];
                int ny = y + dy[d];

                // Ra ngoài lưới
                if (nx < 0 || nx >= n ||
                    ny < 0 || ny >= m) {
                    continue;
                }

                // Đã thăm
                if (visited[nx][ny]) {
                    continue;
                }

                // Ô có độ cao <= mực nước đã bị ngập
                if (height[nx][ny] <= water) {
                    continue;
                }

                visited[nx][ny] = true;
                queue.offer(new int[]{nx, ny});
            }
        }
    }
}

package review;

import java.util.Scanner;

public class NuocBien {

    static int n, m, maxHeight, ans;
    static int[][] a;

    static int[][] d = {
        {0, 1},
        {0, -1},
        {-1, 0},
        {1, 0}
    };

    static class Point {
        int x;
        int y;

        Point(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }

    // Tìm tất cả các ô bị biển ngập
    static void findSea(int height, boolean[][] sea) {

        Point[] q = new Point[n * m];

        int front = 0;
        int rear = 0;

        // Đưa tất cả ô biên vào queue nếu chúng thấp hơn mực nước
        for (int i = 0; i < n; i++) {

            if (a[i][0] <= height && !sea[i][0]) {
                sea[i][0] = true;
                q[rear++] = new Point(i, 0);
            }

            if (a[i][m - 1] <= height && !sea[i][m - 1]) {
                sea[i][m - 1] = true;
                q[rear++] = new Point(i, m - 1);
            }
        }

        for (int j = 0; j < m; j++) {

            if (a[0][j] <= height && !sea[0][j]) {
                sea[0][j] = true;
                q[rear++] = new Point(0, j);
            }

            if (a[n - 1][j] <= height && !sea[n - 1][j]) {
                sea[n - 1][j] = true;
                q[rear++] = new Point(n - 1, j);
            }
        }

        // BFS từ biển
        while (front < rear) {

            Point cur = q[front++];

            for (int[] c : d) {

                int nx = cur.x + c[0];
                int ny = cur.y + c[1];

                if (nx < 0 || nx >= n ||
                    ny < 0 || ny >= m ||
                    sea[nx][ny]) {
                    continue;
                }

                if (a[nx][ny] > height) {
                    continue;
                }

                sea[nx][ny] = true;
                q[rear++] = new Point(nx, ny);
            }
        }
    }

    // Kiểm tra đảo có bị chia không
    static boolean isSplit(int height) {

        boolean[][] sea = new boolean[n][m];

        // Bước 1: tìm những ô thực sự là biển
        findSea(height, sea);

        boolean[][] visited = new boolean[n][m];

        int cnt = 0;

        // Bước 2: đếm số vùng đất còn lại
        for (int i = 0; i < n; i++) {

            for (int j = 0; j < m; j++) {

                // Không phải biển và chưa thăm
                if (!sea[i][j] && !visited[i][j]) {

                    cnt++;

                    // Có từ 2 vùng đất trở lên
                    if (cnt >= 2) {
                        return true;
                    }

                    bfsLand(i, j, sea, visited);
                }
            }
        }

        return false;
    }

    // BFS tìm một vùng đất
    static void bfsLand(
            int x,
            int y,
            boolean[][] sea,
            boolean[][] visited) {

        Point[] q = new Point[n * m];

        int front = 0;
        int rear = 0;

        q[rear++] = new Point(x, y);
        visited[x][y] = true;

        while (front < rear) {

            Point cur = q[front++];

            for (int[] c : d) {

                int nx = cur.x + c[0];
                int ny = cur.y + c[1];

                if (nx < 0 || nx >= n ||
                    ny < 0 || ny >= m) {
                    continue;
                }

                if (visited[nx][ny] || sea[nx][ny]) {
                    continue;
                }

                visited[nx][ny] = true;
                q[rear++] = new Point(nx, ny);
            }
        }
    }

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int t = 1;

        while (true) {

            n = sc.nextInt();
            m = sc.nextInt();

            if (n == 0 && m == 0) {
                break;
            }

            a = new int[n][m];

            maxHeight = 0;

            for (int i = 0; i < n; i++) {

                for (int j = 0; j < m; j++) {

                    a[i][j] = sc.nextInt();

                    maxHeight = Math.max(
                        maxHeight,
                        a[i][j]
                    );
                }
            }

            ans = -1;

            for (int height = 1;
                 height <= maxHeight;
                 height++) {

                if (isSplit(height)) {
                    ans = height;
                    break;
                }
            }

            if (ans == -1) {

                System.out.println(
                    "Case " + t +
                    ": Island never splits."
                );

            } else {

                System.out.println(
                    "Case " + t +
                    ": Island splits when ocean rises " +
                    ans + " feet."
                );
            }

            t++;
        }

        sc.close();
    }
}

	};

		}

		Point[] q = new Point[n*m];
		
		int front = 0;
		int rear = 0;
		
		for ( int i=0; i<n; i++){
			if ( a[i][0] <= height && !sea[i][0]){

				q[rear++] = new Point(i, m-1);
			}

		for ( int j = 0; j<m; j++){

				q[rear++] = new Point(0, j);

				sea[nx][ny] = true;

			}
		}
	}
	
	
	static void bfs(int x, int y, boolean[][] sea, boolean[][] visited){

		int front = 0;
