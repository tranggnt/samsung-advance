package review;

import java.util.Scanner;

public class Partition {

	public static void main(String[] agrs){
		Scanner sc = new Scanner(System.in);
		int tc = sc.nextInt();
		for ( int t = 1; t<=tc; t++){
			int n = sc.nextInt();
			int[] a = new int[n];
			int sum = 0;
			for ( int i =0; i<n; i++){
				a[i] = sc.nextInt();
			}
			for ( int st = 0; st<n-1; st++){
				int min1 = Integer.MAX_VALUE;
				int min2 = Integer.MAX_VALUE;
				int idx1 = -1;
				int idx2 = -1;
				for ( int i =0; i<n; i++){
					if ( a[i] == -1 ){
						continue;
					}
					if ( a[i] < min1){
						min2 = min1;
						idx2 = idx1;
						
						min1 = a[i];
						idx1 = i;
					}else if (a[i] < min2){
						min2 = a[i];
						idx2 = i;
					}
				}
				
				int newVal = min1 + min2;
				sum += newVal;
				a[idx1] = -1;
				a[idx2] = newVal;
			}
			System.out.println("Case #" + t);
			System.out.println(sum);
		}
	}
}
