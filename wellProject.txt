package review;

import java.util.Scanner;

public class WellProject {

	public static boolean[] visited;
	
	
	public static int Prim(int n, int[][] a ){
		
		int ans = 0;
		visited[0] = true;
		int count = 1;
		int jmin = -1;
		while ( count < n){
			int min = Integer.MAX_VALUE;
			
			for ( int i = 0; i<n; i++){
				if ( visited[i] == true){
					for ( int j=0; j<n; j++){
						if ( j!=i && visited[j] == false){
							if ( a[i][j] < min){
								min = a[i][j];
								jmin = j;
							}
						}
					}
				}
			}
			visited[jmin] = true;
			ans += min;
			count++;
		}
		return ans;
	}
	
	public static void main(String agrs[]){
		Scanner sc = new Scanner(System.in);
		
		int tc = sc.nextInt();
		for ( int t = 1; t<=tc; t++){
			int n = sc.nextInt();
			int[][] a = new int [ n+5][n+5];
			visited = new boolean[n+5];
			for ( int i =0; i<n; i++){
				visited[i] = false;
				for ( int j = 0; j<n; j++){
					a[i][j] = sc.nextInt();
				}
			}
			int res = Prim(n, a);
			System.out.println("Case #" + t);
			System.out.println(res);			
		}
	}
}
