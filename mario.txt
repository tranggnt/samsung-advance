package review;

import java.util.Scanner;

import review.Princess.Pair;

public class Mario {

	static int[][] d= {
		{1, 0},
		{-1, 0},
		{0, 1},
		{0, -1},
	};
	
	static boolean[][] visited;
	public static class Pair{
		private int x;
		private int y;
		
		
		public Pair(int x, int y){
			this.x = x;
			this.y = y;
			
		}
	}
	
	public static class Queue{
		Pair[] q;
		int front;
		int rear;
		
		public Queue(){
			q = new Pair[1000000];
			front = 0;
			rear  = -1;
		}
		
		public Pair pop(){
			return q[front++];
		}
		
		public boolean isEmpty(){
			return rear < front;
		}
		
		public void push(Pair p){
			q[++rear] = p;
		}
		
		 
	}
	
	static int x1=-1, y1=-1, x2=-1, y2=-1;
	static int n, m;
	static int[][] a;
	
	public static boolean check ( int step){
		for (int i =0; i<n; i++){
			for ( int j=0; j<m; j++){
				visited[i][j] = false;
			}
		}
		Queue q = new Queue();
		Pair p = new Pair(x1, y1);
		q.push(p);
		visited[x1][y1] = true;
		while (!q.isEmpty()){
			Pair tmp = q.pop();
			
			for ( int i=1; i<=step; i++){
				int[] dx = {-i, i, 0, 0};
				int[] dy = {0, 0, -1, 1};
				
				for ( int j =0; j<4; j++){
					int nx = tmp.x + dx[j];
					int ny = tmp.y + dy[j];
					
					if (nx == x2 && ny == y2){
						return true;
					}
					
					if ( nx>=0 && nx < n && ny >=0 && ny <m && a[nx][ny] != 0 && !visited[nx][ny]){
						Pair np = new Pair(nx, ny);
						q.push(np);
						visited[nx][ny] = true;
					}
				}
			}
		}
		return false;
		
	}
	
	public static void main(String[] agrs){
		Scanner sc = new Scanner(System.in);
		int tc = sc.nextInt();
		for ( int t = 1; t<=tc; t++){
			n = sc.nextInt();
			m = sc.nextInt();
			a = new int[n+5][m+5];
			visited = new boolean[n+5][m+5];
			
			
 			for ( int i =0; i<n; i++){
				for ( int j=0; j<m; j++){
					a[i][j] = sc.nextInt();
					visited[i][j] = false;
					if ( a[i][j] == 2){
						x1 = i;
						y1= j;
					} else if ( a[i][j] == 3) {
						x2 = i;
						y2 = j;
					}
				}
			}
 			
 			int ans = -1;
 			for ( int i=1; i<=n ;i++){
 				if (check(i)){
 					ans = i;
 					break;
 				}
 			}
 			System.out.println("Case #" + t);
				System.out.println(ans);
 			
 			
		}
	}
}
