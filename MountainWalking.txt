Cho một bản đồ kích thước NxN (2 <= N <= 100), mỗi ô mang giá trị là độ cao của ô đó (0 <= độ cao <= 110). Bác John và bò Bessie đang ở ô trên trái (dòng 1, cột 1) và muốn đi đến cabin (dòng N, cột N). Họ có thể đi sang phải, trái, lên trên và xuống dưới nhưng không thể đi theo đường chéo. Hãy giúp bác John và bò Bessie tìm đường đi sao cho chênh lệch giữa điểm cao nhất và thấp nhất trên đường đi là nhỏ nhất.

 

Input

Dòng 1: Số test case

Dòng 2: N

N dòng tiếp theo chứa N số nguyên, mỗi số cho biết cao độ của một ô.

 

Output

In ra #test case và một số nguyên là chênh lệch độ cao nhỏ nhất.

package review;

import java.util.Scanner;

public class MountainWalking {
	
	public static class Point{
		int x; int y;
		
		Point(int x, int y){
			this.x = x;
			this.y = y;
		}
	}
	
	static int n;
	static int[][] a;
	static int[][] d = {
		{1, 0},
		{-1, 0},
		{0, 1},
		{0, -1}
	};
	
	
	public static boolean bfs( int low, int high){
		if ( a[0][0] < low || a[0][0] > high || a[n-1][n-1] < low || a[n-1][n-1] > high){
			return false;
		}
		
		boolean[][] visited = new boolean[n+5][n+5];
		
		Point[] q = new Point[n*n];
		int front = 0;
		int rear = 0;
		visited[0][0] = true;
		q[rear] = new Point(0, 0);
		rear ++;
		while ( front < rear){
			Point cur = q[front];
			front++;
			for ( int c[] : d){
				int nx = cur.x + c[0];
				int ny = cur.y + c[1];
				
				if ( nx == n-1 && ny == n-1){
					return true;
				}
				
				if ( nx >=0 && nx < n && ny >=0 && ny <n && !visited[nx][ny]){
					if ( a[nx][ny] >= low && a[nx][ny] <= high){
						q[rear] = new Point(nx, ny);
						rear++;
						visited[nx][ny] = true;
					}
					
				}
			}
		}
		return false;
	}
	
	public static void main(String[] args){
		
		Scanner sc = new Scanner(System.in);
		int tc = sc.nextInt();
		for ( int t =1; t<=tc; t++){
			n = sc.nextInt();
			a = new int [n+5][n+5];
			
			for ( int i=0; i<n; i++){
				for ( int j=0; j<n; j++){
					a[i][j] = sc.nextInt();
				}
			}
			int ans = Integer.MAX_VALUE;
			int low = 0; int high = 0;
			while ( low <=100 && high <= 110){
				if ( bfs(low, high)){
					ans = Math.min(ans, (high-low));
					low++;
				}else{
					high++;
				}
			}
			
			System.out.println("#" + t + " " + ans);
		}
		
	}
}
