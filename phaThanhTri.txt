package review;

import java.util.Scanner;

public class PhaThanhTri {
	
	static int[][] e;
	static int m;
	static boolean[] visited;
	
	public static int maxSpanningTree(){
		visited[0] = true;
		int count = 1;
		int jmax = -1;
		int ans = 0;
		while ( count < m){
			int max = Integer.MIN_VALUE;
			for ( int i = 0; i<m; i++){
				if ( visited[i]){
					for ( int j =0; j<m; j++){
						if ( !visited[j] && j!=i){
							if ( e[i][j] > max){
								max = e[i][j];
								jmax = j;
							}
						}
					}
				}
			}
			visited[jmax] = true;
			ans += max;
			count++;
		}
		return ans;
	}

	public static void main(String[] agrs){
		Scanner sc = new Scanner(System.in);
		
		int tc = sc.nextInt();
		for ( int t = 1; t<=tc; t++){
			m = sc.nextInt();
			int[] a = new int[m+5];
			e = new int[m+5][m+5];
			visited = new boolean[m+5];
			for ( int i =0; i<m; i++){
				visited[i] = false;
				int id = sc.nextInt(); 
				a[id] = sc.nextInt(); 
				int k = sc.nextInt();
				for ( int j = 0; j<k; j++){
					int c = sc.nextInt();
					e[id][c] = 1;
				}
			}
			

			int sum = 0;
			for ( int i =0; i<m; i++){
				for ( int j=0; j<m; j++){
					if ( e[i][j] == 1){
						e[i][j] = a[i] + a[j];
						sum += e[i][j];
					}
				}
				
			}
			


			int res = sum/2 - maxSpanningTree();
			System.out.println("Case #" + t);
			System.out.println(res);
		}
	}
}
