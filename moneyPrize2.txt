package review;

import java.util.Scanner;

public class MoneyPrize2 {
	
	static int[] count = new int[20];
	static String[][] store = new String[20][1000];
	
	public static void clear(){
		for ( int i=0; i<20; i++){
			count[i] = 0;
			for ( int j= 0; j<1000; j++){
				store[i][j] = null;
			}
		}
	}
	
	public static void swap(char[] num, int a, int b ){
		char tmp = num[a];
		num[a] = num[b];
		num[b] = tmp;
	}
	
	public static void insert(String num, int step){
		for ( int i = 0; i<count[step]; i++){
			if ( store[step][i].equals(num)){
				return;
			}
		}
		store[step][count[step]] = num;
		count[step]++;
		
	}
	
	public static void generate(String num, int step){
		char[] a = num.toCharArray();
		
		for ( int i=0; i<a.length-1; i++){
			for ( int j = i+1; j<a.length; j++){
				swap(a,i, j);
				String newNum = new String(a);
				insert(newNum, step);
				swap(a,i,j);
			}
		}
	}
	
	
	public static void main(String agrs[]){
		Scanner sc = new Scanner(System.in);
		int tc = sc.nextInt();
		sc.nextLine();
		for ( int t =1; t<=tc; t++){
			
			String s = sc.nextLine();
			String[] ss = s.split("\\s+");
			int k = Integer.valueOf(ss[1]);
			clear();
			store[0][0] = ss[0];
			count[0] = 1;
			
			for ( int step =1; step<=k; step++){
				for ( int i = 0; i<count[step-1]; i++){
					String num = store[step-1][i];
					generate(num, step);
				}
			}
			
//			String res = store[k][0];
//			for ( int i = 0; i<count[k]; i++){
//				if (store[k][i].compareTo(res) > 0){
//					res = store[k][i];
//				}
//			}
			
			int res = Integer.valueOf(store[k][0]);
			
			for ( int i =0; i<count[k]; i++){
				int n = Integer.valueOf(store[k][i]);
				if ( n > res) res = n;
			}
			
			System.out.println("Case #" + t);
			System.out.println(res);
			
		}
	}

}
