package review;

import java.util.Scanner;

public class CutPaper {

	
	public static boolean check ( int x, int y, int n, int [][] a){
		int c = a[x][y];
		for (int i =x; i<x+n; i++){
			for ( int j = y; j< y+n; j++){
				if ( a[i][j] != c) return false; 
			}
		}
		return true;
	}
	
	
	static int white = 0, blue = 0;
	
	public static void Try (int x, int y, int n, int[][] a){
		
		
		if ( check (x, y, n, a)){
			if (a[x][y] == 1 ) blue ++;
			else white++;
		}
		else {
			
			Try(x, y, n/2, a);
			Try(x+n/2, y, n/2, a);
			Try(x, y+n/2 , n/2, a);
			Try(x+n/2, y+n/2, n/2, a);
		}	
	}
	
	public static void main(String agrs[]){
		
		Scanner sc = new Scanner(System.in);
		
		int tc = sc.nextInt();
		for ( int t= 1; t<=tc; t++){
			int n = sc.nextInt();
			int [][]  a = new int[1000][1000];
			for ( int i = 0 ; i < n ; i++){
				for ( int j =0; j<n; j++){
					a[i][j] = sc.nextInt();
				}
			}
			
			white = 0;
			blue = 0;
			
			Try(0, 0 , n, a);
			System.out.println("Case #" + t);
			System.out.println(white + " " + blue);
		}
	}
}
