package review;

import java.util.Scanner;

public class PickingUpJewels {
	
	static int n;
	static int[][] a;
	static boolean[][] visited;
	
	static int max;
	static int[][] d = {
		{1, 0},
		{-1, 0},
		{0, 1},
		{0, -1}
	};
	
	public static void Try(int i, int j, int score){
		if ( i<0 || i>n-1 || j<0 || j>n-1 || a[i][j] == 1|| visited[i][j]){
			return;
		}
		if ( a[i][j] == 2){
			score+=1;
		}
		visited[i][j] = true;
		
		if (i==n-1 && j==n-1){
			if ( max < score){
				max = score;
			}
			visited[i][j] = false;
			return;
		}
		
		for ( int c[] : d){
			Try(i+c[0], j+c[1], score);
		}
		visited[i][j] = false;
	}
	public static void main(String[] args){
		Scanner sc = new Scanner(System.in);
		int tc = sc.nextInt();
		for ( int t=0; t<tc; t++){
			n = sc.nextInt();
			a = new int[n][n];
			visited = new boolean[n+5][n+5];
			for ( int i=0; i<n; i++){
				for ( int j =0; j<n; j++){
					a[i][j] = sc.nextInt();
					visited[i][j] = false;
				}
			}
			max = Integer.MIN_VALUE;
			Try(0,0,0);
			System.out.println(max);
		}
	}
}
