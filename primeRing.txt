package review;

import java.util.Scanner;

public class PrimeRing {
	
	static int n;
	static int[] a;
	static int[] used ;
	static int[] x;
	static int cnt;
	
	
	static boolean isPrime( int n){
		if ( n < 2) return false;
		for ( int i =2; i<=Math.sqrt(n); i++){
			if ( n%i == 0) return false;
		}
		return true;
	}
	
	static int[] ngto = new int [300];
	

	
	static void Try ( int i){
		if ( i == n+1){
			int c = x[n] + x[1];
			if ( ngto[c] == 1){
				cnt++;
			}
			return;
		}
		for ( int j=2; j<=n; j++){
			if ( used[j] == 0){
				int b = x[i-1] + a[j];
				if ( ngto[b] == 0){
					continue;
				}
				used[j] = 1;
				x[i] = a[j];
				Try(i+1);
				used[j] = 0;
			}
		}
		
		
	}
	
	public static void main(String[] args ){
		Scanner sc = new Scanner(System.in);
		int tc =sc.nextInt();
		
		ngto[0]=ngto[1]=0;
		for ( int i=2; i<=200; i++){
			if ( isPrime(i)){
				ngto[i] = 1;
			}
		}
		
		for ( int t = 1; t<=tc ;t++){
			n = sc.nextInt();
			a = new int[n+5];
			cnt = 0;
			x = new int[n+5];
			used = new int [n+5];
			
			for ( int i=1; i<=n; i++){
				a[i] = sc.nextInt();
				used[i] =  0;
			}
			
			used[1] = 1;
			x[1] = a[1];
			Try(2);
			System.out.println("Case " + t + ": " + cnt);
		}
	}
}
