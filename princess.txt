package review;

import java.util.Scanner;

public class Princess {

	
	static int[][] d = {
		{0, 1},
		{0, -1},
		{1, 0},
		{-1, 0}
	};
	
	public static class Pair{
		int x;
		int y;
		int step;
		
		public Pair(int x, int y, int step){
			this.x  =x ;
			this.y = y; 
			this.step = step;
		}
	}
	
	
	public static class Queue{
		public Pair[] q;
		int front;
		int rear;
		
		public Queue(){
			q = new Pair[1000000];
			front = 0;
			rear  = -1;
		}
		
		public void push ( Pair p){
			q[++rear] = p;
		}
		
		public Pair pop(){
			return 	q[front++];
		}
		
		public boolean isEmpty(){
			return rear < front;
		}
		
	}
	
	static boolean[][] visited;
	
	
	public static void main(String[] agrs){
		Scanner sc = new Scanner(System.in);
		int tc = sc.nextInt();
		for ( int t =0; t<tc; t++){
			int n = sc.nextInt();
			int[][] a = new int[n+5][n+5];
			visited = new boolean[n+5][n+5];
			//int pX = -1, pY = -1;
			for ( int i=1; i<=n; i++){
				for ( int j = 1; j<=n; j++){
					a[i][j] = sc.nextInt();
					visited[i][j] = false;
				}
			}
			boolean found = false;
			Queue q = new Queue();
			Pair p = new Pair(1, 1, 0);
			q.push(p);
			visited[1][1] = true;
			int step1 = 0;
			while ( !q.isEmpty()){
				if (found) break;
				Pair tmp = q.pop();
				
				for ( int c[] : d){
					int i = tmp.x+c[0];
					int j = tmp.y+c[1];

					
					if ( a[i][j] == 2){
						found = true;
						step1 = tmp.step+1;
						break;
					}
					if ( i >=1 && i<=n && j>=1 && j<=n && !visited[i][j] && a[i][j] == 1){
						Pair pp = new Pair(i, j, tmp.step+1);
						visited[i][j] = true;
						q.push(pp);
					}
					
				}
			}
			
			if ( found == false ){
				System.out.println(-1);
			}
			else{
				for ( int i=1; i<=n; i++){
					for ( int j = 1; j<=n; j++){
						visited[i][j] = false;
					}
				}
				Queue q2 = new Queue();
				visited[n][n] = true;
				boolean exit =false;
				int step2 = 0;
				Pair pa = new Pair(n, n, 0);
				q2.push(pa);
				while ( !q2.isEmpty()){
					if ( exit){
						break;
					}
					Pair tmp = q2.pop();
					for ( int c[] : d){
						int i = tmp.x + c[0];
						int j = tmp.y + c[1];
						if ( a[i][j] == 2){
							exit = true;
							step2 = tmp.step+1;
							break;
						}
						if ( i >=1 && i<=n && j>=1 && j<=n && !visited[i][j] && a[i][j] == 1){
							Pair pp = new Pair(i, j, tmp.step+1);
							visited[i][j] = true;
							q2.push(pp);
						}
					}
				}
				//System.out.println("OLA" + step2);
				if ( exit){
					System.out.println(step1 + step2);
				}else {
					System.out.println(-1);
				}
			}
			
			
		}
	}
}
