package review;

import java.util.Scanner;

public class TheFrog {

	public static class Point{
		int x;
		int y;
		int r;
		
		public Point(int x, int y, int r){
			this.x = x;
			this.y=y;
			this.r = r;
		}
		
	}
	
	public static double dis(Point a, Point b){
		double d = (a.x-b.x) * (a.x-b.x) + (a.y-b.y) * (a.y-b.y);
		double distance = Math.sqrt(d)-a.r-b.r;
		return distance;
	}
	
	public static void main(String[] agrs){
		Scanner sc = new Scanner(System.in);
		
		int tc = sc.nextInt();
		for ( int t = 1; t<=tc; t++){
			int n =sc.nextInt();
			Point[] p = new Point[n];
			for ( int i =0; i<n; i++){
				int x = sc.nextInt();
				int y = sc.nextInt();
				int r = sc.nextInt();
				p[i] = new Point(x, y, r);
			}
			
			int[][] a = new int [n+5][n+5];
			
			for ( int i =0; i< n; i++){
				for ( int j = i+1; j<n; j++){
					if ( dis(p[i], p[j]) <= 40.0){
						a[i][j] = 1;
						a[j][i] = 1;
					}
					else if ( dis(p[i], p[j]) > 40.0 && dis(p[i], p[j]) <=90.0 ){
						a[i][j] = 200;
						a[j][i] = 200;
					}
					else if (dis(p[i], p[j]) > 90.0 ){
						a[i][j] = Integer.MAX_VALUE;
						a[j][i] = Integer.MAX_VALUE;
					}
				}
			}
			
			boolean[] visited = new boolean[n];
			int[] d = new int [n];
			for ( int i =0; i<n; i++){
				d[i] = Integer.MAX_VALUE;
			}
			d[0] = 0;
			
			for ( int cnt = 0; cnt < n; cnt++){
				int u = -1 ;
				int min = Integer.MAX_VALUE;
				
				for ( int i =0; i<n; i++){
					if ( !visited[i] && d[i] < min){
						min = d[i];
						u = i;
					}
				}
				
				if ( u == -1) break;
				visited[u] = true;
				
				for ( int j=0; j<n; j++){
					if ( !visited[j] && a[u][j] !=0 && a[u][j] != Integer.MAX_VALUE){
						if ( d[j] > d[u] + a[u][j]){
							d[j] = d[u] + a[u][j];
						}
					}
				}
			}
			if ( d[n-1] == Integer.MAX_VALUE){
				System.out.println(-1);
			} else {
				int xa = d[n-1] / 200;
				int gan = d[n-1] % 200;
				System.out.println(xa + " " + gan);
			}
		}
	}
}
