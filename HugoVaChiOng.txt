Hugo và chị ong nâu nấu nâu nấu lầu nâu
Hugo và chị ong nâu là hàng xóm của nhau. Tuy nhiên Hugo thì lười nhưng lại muốn lấy mật ăn, còn chị ong nâu thì chăm chỉ kiếm mật để đầy tổ của mình. Vì thèm ăn mật ong mà mãi chị ong nâu không cho nên sau 1 thời gian nằm gai nếm mật rình mò Hugo đã phát hiện ra các bố trí của các thùng chứa mật và số mật trong mỗi thùng ở nhà chị ong nâu. Mật ở nhà chị ong nâu được biểu diễn bằng 1 ma trận NxM ô, mỗi ô một thùng mật. Mỗi ô chứa lượng mật nhất định và có thể liên kết với 6 ô xung quanh theo cách bố trí của tổ ong. 2 ô được gọi là có liên kết nếu chúng có chung cạnh.

Hôm nay Hugo phát hiện chị ong nâu đi sang khu rừng bên cạnh để lấy mật. Hugo sẽ vào nhà chị ong nâu để ăn trộm mật ong của chị. Để không bị phát hiện Hugo chỉ có thể lấy tối đa 4 thùng ở 4 ô có liên kết với nhau. Hãy giúp Hugo tìm ra bình phương tổng lượng mật lớn nhất có thể lấy.

Ex: M = 5 , N = 3



Trong ví dụ trên, bình phương tổng lượng mật là

(300 + 410 + 185 + 95)2 = 980100

 



Trong ví dụ trên Hugo sẽ bị phát hiện và không thể lấy mật.

 

[Input]

import java.util.Scanner;

public class Main {
    static int N, M;
    static long maxHoney;
    static int[][] grid;
    static boolean[][] visited;

    // Hướng di chuyển cho hàng chẵn (index 0-based: 0, 2, 4...)
    static int[][] dirEven = {
        {0, -1}, {0, 1},     // Trái, Phải
        {-1, -1}, {-1, 0},   // Trên-Trái, Trên-Phải
        {1, -1}, {1, 0}      // Dưới-Trái, Dưới-Phải
    };

    // Hướng di chuyển cho hàng lẻ (index 0-based: 1, 3, 5...)
    static int[][] dirOdd = {
        {0, -1}, {0, 1},     // Trái, Phải
        {-1, 0}, {-1, 1},    // Trên-Trái, Trên-Phải
        {1, 0}, {1, 1}       // Dưới-Trái, Dưới-Phải
    };

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        if (!sc.hasNextInt()) return;

        int T = sc.nextInt();
        for (int tc = 1; tc <= T; tc++) {
            M = sc.nextInt(); // Số cột
            N = sc.nextInt(); // Số hàng

            grid = new int[N][M];
            visited = new boolean[N][M];

            for (int i = 0; i < N; i++) {
                for (int j = 0; j < M; j++) {
                    grid[i][j] = sc.nextInt();
                }
            }

            maxHoney = 0;

            // Thử bắt đầu từ mọi ô trong ma trận
            for (int r = 0; r < N; r++) {
                for (int c = 0; c < M; c++) {
                    visited[r][c] = true;
                    dfs(r, c, 1, grid[r][c]);
                    visited[r][c] = false;
                }
            }

            // Kết quả là Bình phương tổng lượng mật lớn nhất
            long result = maxHoney * maxHoney;
            System.out.println("Case #" + tc);
            System.out.println(result);
        }
        sc.close();
    }

    static void dfs(int r, int c, int count, long currentSum) {
        // Khi chọn đủ 4 ô liên kết
        if (count == 4) {
            if (currentSum > maxHoney) {
                maxHoney = currentSum;
            }
            return;
        }

        // Chọn tập các hướng di chuyển tùy theo hàng chẵn hay lẻ
        int[][] dirs = (r % 2 == 0) ? dirEven : dirOdd;

        for (int[] dir : dirs) {
            int nr = r + dir[0];
            int nc = c + dir[1];

            // Kiểm tra ô hợp lệ trong ma trận và chưa được thăm
            if (nr >= 0 && nr < N && nc >= 0 && nc < M && !visited[nr][nc]) {
                visited[nr][nc] = true;
                dfs(nr, nc, count + 1, currentSum + grid[nr][nc]);
                visited[nr][nc] = false; // Backtrack
            }
        }
    }
}
